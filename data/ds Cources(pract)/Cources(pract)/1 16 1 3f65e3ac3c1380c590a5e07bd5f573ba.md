Inside `returnCard` function 

```cpp
int index = hashFunction(bookId);
    Card* current = hashTable[index];
    Card* prev = nullptr;
    
    while (current != nullptr) {
        if (current->bookId == bookId) {
            if (prev == nullptr) {
                hashTable[index] = current->next;
            } else {
                prev->next = current->next;
            }
            delete current;
            cout << "Card removed for Book ID " << bookId << "\n";
            return;
        }
        prev = current;
        current = current->next;
    }
    cout << "Card not found for Book ID " << bookId << "\n";
```
