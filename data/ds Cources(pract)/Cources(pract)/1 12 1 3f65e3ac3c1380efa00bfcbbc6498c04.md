        }
    }

    for (int i = 0; i < edges; i++) {
        int u, v;
        while (true) {
            cout << "Edge: ";
            cin >> u >> v;
            if (u >= 0 && u < n && v >= 0 && v < n) {
                adj[u][v] = 1;
                adj[v][u] = 1;
                break;
            } else {
                cout << "Invalid edge\n";
            }
        }
    }

    queue[rear++] = start;
    bfsVisited[start] = 1;

    while (front < rear) {
        int current = queue[front++];
        cout << current << " ";

        for (int i = 0; i < n; i++) {
            if (adj[current][i] == 1 && !bfsVisited[i]) {
                queue[rear++] = i;
                bfsVisited[i] = 1;
            }
        }
    }
