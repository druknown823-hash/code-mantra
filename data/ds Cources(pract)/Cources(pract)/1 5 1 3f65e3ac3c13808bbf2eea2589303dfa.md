void display

```cpp
if (top == -1) {
        cout << "Stack is empty\n";
    } else {
        cout << "Stack elements:";
        for (int i = 0; i <= top; i++) {
            cout << " " << stack[i];
        }
        cout << " " << "\n";
    }
```

```cpp
if (top == MAX - 1) {
```

void pop

```cpp
if (top == -1) {
        cout << "Stack Underflow\n";
    } else {
        cout << stack[top] << " popped from stack\n";
        top--;
    }
```
