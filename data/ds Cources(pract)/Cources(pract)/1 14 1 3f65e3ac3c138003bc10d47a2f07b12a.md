    while (qFront < qRear) {
        Node curr = queue[qFront++];

        if (curr.moves >= 7) continue;

        for (int m = 0; m < 6; m++) {
            Node next = curr;
            next.moves = curr.moves + 1;
            applyMove(next.state, m);

            if (isSolved(next.state)) {
                return next.moves;
            }

            if (next.moves < 7 && qRear < MAX_STATES) {
                queue[qRear++] = next;
            }
        }
    }

    return -1;
