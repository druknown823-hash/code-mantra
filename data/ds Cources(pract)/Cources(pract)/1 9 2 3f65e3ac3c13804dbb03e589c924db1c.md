if (*head == NULL || pos < 0) {

        if (temp == NULL) break;
        temp = temp->next;

        cout << "Position is out of bounds" << endl;
        return;
    }

    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->data = value;
