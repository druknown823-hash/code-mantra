Under `#define MAX 10`

```cpp
int n;
int dist[MAX][MAX];
bool visited[MAX];
int minCost = INT_MAX;
int currentPath[MAX + 1];
int bestPath[MAX + 1];
```

Inside `tsp` function 

```cpp
if (count == n && dist[currPos][0] > 0) {
        int totalCost = cost + dist[currPos][0];
        if (totalCost < minCost) {
            minCost = totalCost;
            for (int i = 0; i < n; i++) {
                bestPath[i] = currentPath[i];
            }
            bestPath[n] = 0;
        }
        return;
    }

    for (int i = 0; i < n; i++) {
        if (!visited[i] && dist[currPos][i] > 0) {
            visited[i] = true;
            currentPath[count] = i;
            
            tsp(i, count + 1, cost + dist[currPos][i]);
            
            visited[i] = false;
        }
    }
```
