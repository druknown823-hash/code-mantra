Inside `greedyTravelPlan` function 

```cpp
cout << "Greedy Travel Plan (Nearest Neighbour):\n";
    bool visited[MAX_CITIES] = {false};
    int curr = 0;
    visited[curr] = true;
    int totalDist = 0;

    cout << cities[curr];

    for (int i = 0; i < MAX_CITIES - 1; i++) {
        int nextCity = -1;
        int minD = INF;

        for (int j = 0; j < MAX_CITIES; j++) {
            if (!visited[j] && dist[curr][j] < minD) {
                minD = dist[curr][j];
                nextCity = j;
            }
        }

        visited[nextCity] = true;
        totalDist += minD;
        cout << " -> " << cities[nextCity];
        curr = nextCity;
    }

    totalDist += dist[curr][0];
    cout << " -> " << cities[0] << "\n";
    cout << "Total Distance: " << totalDist << " km\n";
```
