void display

```cpp
if (front == -1 || front > rear) {
        cout << "Queue is empty\n";
    } else {
        cout << "Queue elements: ";
        for (int i = front; i <= rear; i++) {

        }
        cout << "\n";
    }
```

```cpp
if (rear == MAX - 1) {
        cout << "Queue Overflow\n";

        if (front == -1) {
```

void dequeue

```cpp
if (front == -1 || front > rear) {
        cout << "Queue Underflow\n";
    } else {
        cout << queue[front] << " deleted from queue\n";
        front++;
        // Reset queue to empty state if all elements are deleted
        if (front > rear) {
            front = -1;
            rear = -1;
        }
    }
```
